### Задание 2. DI-контейнер и сервисы (≈50 минут)

**Шаг 1.** Создайте контейнер в `src/shared/di/container.ts`:

```ts
type Factory<T> = () => T;

/** Минималистичный DI-контейнер: регистрация фабрик + ленивые синглтоны. */
export class Container {
  private readonly factories = new Map<string, Factory<unknown>>();
  private readonly instances = new Map<string, unknown>();

  register<T>(token: string, factory: Factory<T>): void {
    this.factories.set(token, factory);
    this.instances.delete(token); // старое изделие больше не действительно
  }

  resolve<T>(token: string): T {
    const cached = this.instances.get(token);
    if (cached !== undefined) {
      return cached as T;
    }

    const factory = this.factories.get(token);
    if (factory === undefined) {
      throw new Error(`В контейнере не зарегистрирована зависимость "${token}"`);
    }

    const instance = factory();
    this.instances.set(token, instance);
    return instance as T;
  }

  clear(): void {
    this.instances.clear();
  }
}

export const TOKENS = {
  httpClient: 'httpClient',
  tokenStorage: 'tokenStorage',
  authService: 'authService',
} as const;
```

**Шаг 2.** Создайте слой данных: `src/shared/api/http-client.ts`:

```ts
export interface HttpClient {
  get<T>(url: string): Promise<T>;
  post<T>(url: string, body: unknown): Promise<T>;
}

export class FetchHttpClient implements HttpClient {
  constructor(private readonly baseUrl: string) {}

  async get<T>(url: string): Promise<T> {
    const response = await fetch(`${this.baseUrl}${url}`);
    if (!response.ok) {
      throw new Error(`HTTP ${response.status}: ${response.statusText}`);
    }
    return (await response.json()) as T;
  }

  async post<T>(url: string, body: unknown): Promise<T> {
    const response = await fetch(`${this.baseUrl}${url}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body),
    });
    if (!response.ok) {
      throw new Error(`HTTP ${response.status}: ${response.statusText}`);
    }
    return (await response.json()) as T;
  }
}
```

**Шаг 3.** Создайте сервисы аутентификации: `src/entities/user/api/auth-service.ts` (или в `features/auth-by-form` — обсудите на занятии, почему место спорное):

```ts
export interface TokenStorage {
  getToken(): string | null;
  setToken(token: string): void;
  clearToken(): void;
}

/** Браузерная реализация: localStorage. */
export class BrowserTokenStorage implements TokenStorage {
  private readonly key = 'app.token';

  getToken(): string | null {
    return localStorage.getItem(this.key);
  }

  setToken(token: string): void {
    localStorage.setItem(this.key, token);
  }

  clearToken(): void {
    localStorage.removeItem(this.key);
  }
}

/** Тестовая реализация: память. Работает и в Node.js. */
export class MockTokenStorage implements TokenStorage {
  private token: string | null = null;

  getToken(): string | null {
    return this.token;
  }

  setToken(token: string): void {
    this.token = token;
  }

  clearToken(): void {
    this.token = null;
  }
}

export class AuthService {
  constructor(
    private readonly http: HttpClient,
    private readonly tokens: TokenStorage,
  ) {}

  async login(email: string, password: string): Promise<void> {
    const response = await this.http.post<{ token: string }>('/auth/login', {
      email,
      password,
    });
    this.tokens.setToken(response.token);
  }

  logout(): void {
    this.tokens.clearToken();
  }

  isAuthenticated(): boolean {
    return this.tokens.getToken() !== null;
  }
}
```

**Шаг 4.** Соберите «корень композиции» — место, где приложение решает, какие реализации использовать. Создайте `src/app/composition-root.ts`:

```ts
import { FetchHttpClient } from '@/shared/api/http-client';
import { Container, TOKENS } from '@/shared/di/container';
import {
  AuthService,
  BrowserTokenStorage,
  MockTokenStorage,
} from '@/entities/user/api/auth-service';

export interface AppEnvironment {
  apiBaseUrl: string;
  emulate?: boolean; // режим разработки/тестов без браузера
}

export function createContainer(env: AppEnvironment): Container {
  const container = new Container();

  container.register(TOKENS.httpClient, () => new FetchHttpClient(env.apiBaseUrl));
  container.register(TOKENS.tokenStorage, () =>
    env.emulate === true ? new MockTokenStorage() : new BrowserTokenStorage(),
  );
  container.register(
    TOKENS.authService,
    () =>
      new AuthService(
        container.resolve<HttpClient>(TOKENS.httpClient),
        container.resolve<TokenStorage>(TOKENS.tokenStorage),
      ),
  );

  return container;
}
```

### Задание 3. Раздача контейнера компонентам (≈20 минут)

Во Vue контейнер раздаётся через `provide`/`inject`. В `src/main.ts`:

```ts
import { createApp, type InjectionKey } from 'vue';

import App from '@/app/ui/App.vue';
import { createContainer } from '@/app/composition-root';
import type { Container } from '@/shared/di/container';

export const DI_CONTAINER: InjectionKey<Container> = Symbol('di-container');

const container = createContainer({
  apiBaseUrl: '/api',
  emulate: import.meta.env.MODE === 'test',
});

createApp(App).provide(DI_CONTAINER, container).mount('#app');
```

В компоненте (задел на тему 9):

```vue
<script setup lang="ts">
import { inject } from 'vue';

import { DI_CONTAINER } from '@/main';
import { TOKENS, type Container } from '@/shared/di/container';
import type { AuthService } from '@/entities/user/api/auth-service';

const container = inject(DI_CONTAINER);
const auth = container?.resolve<AuthService>(TOKENS.authService);
</script>
```

**Ожидаемый результат:** `pnpm build` проходит; компонент получает `AuthService` из контейнера, не зная, как тот собран. Закоммитьте: `feat(di): добавлен контейнер внедрения зависимостей`.
